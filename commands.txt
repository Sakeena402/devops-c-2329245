==================================================
1. GIT SETUP AND GITHUB
==================================================

git init -b main
# Initializes the local Git repository on the main branch.

git status
# Checks the working tree status before staging initial files.

git status --ignored
# Verifies that the evidence/ directory is properly excluded by .gitignore.

git add .gitignore site/
# Stages .gitignore and the initial site/ files.

git commit -m "Initial commit with book catalogue"
# Creates the initial commit containing the catalogue with three books and seat number.

Invoke-RestMethod -Uri "https://api.github.com/user/repos" -Method Post -Body '{"name":"devops-c-2329245","private":false}'
# Creates the public GitHub repository named devops-c-2329245 via GitHub REST API.

git remote add origin https://github.com/Sakeena402/devops-c-2329245.git
# Connects the local repository to the remote GitHub repository.

git branch -M main
# Ensures the primary branch is named main.

git push -u origin main
# Pushes the initial commit to the remote main branch.

==================================================
2. BRANCH AND MERGE
==================================================

git checkout -b update-content
# Creates and checks out a new branch named update-content.

git add site/index.html
# Stages the updated index.html with the fourth book title.

git commit -m "Add fourth book title to catalogue"
# Commits the fourth book to the update-content branch.

git checkout main
# Switches back to the main branch.

git merge update-content --no-ff -m "Merge branch 'update-content' into main"
# Merges update-content into main, creating a clear merge commit.

git push origin main
# Pushes the updated main branch to GitHub.

git push -u origin update-content
# Pushes the update-content branch to GitHub for examiner verification.

git log --graph --oneline --all
# Verifies commit history showing initial commit, update-content commit, and merge commit.

git branch -a
# Verifies that both main and update-content branches exist locally and on the remote.

==================================================
3. GITHUB PAGES
==================================================

Invoke-RestMethod -Uri "https://api.github.com/repos/Sakeena402/devops-c-2329245/pages" -Method Post -Body '{"build_type":"workflow"}'
# Configures repository GitHub Pages source build type to GitHub Actions workflow.

git add .github/workflows/pages.yml
# Stages the GitHub Actions Pages workflow file.

git commit -m "Add GitHub Pages workflow"
# Commits the GitHub Pages workflow file to main.

git push origin main
# Pushes the workflow file to GitHub to trigger the automated deployment.

Invoke-RestMethod -Uri "https://api.github.com/repos/Sakeena402/devops-c-2329245/actions/runs"
# Monitors GitHub Actions workflow runs until the first deployment succeeds.

Invoke-WebRequest -Uri "https://sakeena402.github.io/devops-c-2329245/" -UseBasicParsing
# Verifies the live GitHub Pages website displays the initial catalogue and seat number.

git add site/index.html
# Stages index.html with the updated heading 'College Book Catalogue - Updated'.

git commit -m "Update catalogue heading to College Book Catalogue - Updated"
# Commits the updated heading on main.

git push origin main
# Pushes the heading update to main to trigger the second Pages deployment workflow.

Invoke-RestMethod -Uri "https://api.github.com/repos/Sakeena402/devops-c-2329245/actions/runs"
# Monitors the second GitHub Actions workflow run until completion and success.

Invoke-WebRequest -Uri "https://sakeena402.github.io/devops-c-2329245/" -UseBasicParsing -Headers @{ "Cache-Control" = "no-cache" }
# Verifies that the updated heading is live and visible at the GitHub Pages URL.
